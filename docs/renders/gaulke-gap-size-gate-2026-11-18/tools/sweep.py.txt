import sys,os,json,hashlib,math,collections,pickle
sys.path.insert(0,'.')
from pathlib import Path
from tools import pro_pair, pro_baseline as pb, corpus_scorecard as cs
S='/tmp/claude-0/-home-user-EMB-Bot/82ff0188-f395-586c-97d4-87aeefc800dd/scratchpad/'
FLAG=sys.argv[3] if len(sys.argv)>3 else 'growth_keeps_gaps_between'
shard,n=int(sys.argv[1]),int(sys.argv[2])
from tools.tripwire import committed_jobs
jobs=[('job',s,pb.resolve(pb.JOBS[s][2]),pb.JOBS[s][1],pb.JOBS[s][0]) for s in committed_jobs()]
jobs+=[('fix',f,cs.TESTDATA/f,80.0,'hat_front') for f in cs.FIXTURES]
def stats(d):
    st=d.design['stitches'];c=collections.Counter(s['type'] for s in st)
    nu=0;prev=None
    for s in st:
        if s['type']=='jump' and prev is not None: nu+=math.hypot(s['x']-prev[0],s['y']-prev[1])/10
        if 'x' in s: prev=(s['x'],s['y'])
    runs=collections.Counter(r['kind'] for r in d.design.get('runs',[]))
    shapes=sorted({r['shape'] for r in d.design.get('runs',[]) if r.get('shape')})
    return dict(md5=hashlib.md5(json.dumps(st).encode()).hexdigest(),stitches=c['stitch'],trims=c['trim'],jumps=c['jump'],needle_up_mm=round(nu,1),runs=dict(runs),shapes=shapes,warn=[w.get('code') for w in getattr(d.plan,'warnings',[])])
for i,(kind,name,art,w,g) in enumerate(jobs):
    if i%n!=shard: continue
    out=S+f'sweep/{FLAG}__{kind}__{name.replace("/","_")}.json'
    if os.path.exists(out): continue
    res={'kind':kind,'name':name,'width':w,'garment':g}
    try:
        for tag,ex in (('off',None),('on',{'extra.'+FLAG:True})):
            d=pro_pair.digitize(art,w,g,ex); res[tag]=stats(d)
            if tag=='on' and res['on']['md5']!=res['off']['md5']: pickle.dump(d.design,open(out[:-5]+'_on.pkl','wb'))
            if tag=='off': off_design=d.design
        if res['on']['md5']!=res['off']['md5']: pickle.dump(off_design,open(out[:-5]+'_off.pkl','wb'))
    except Exception as e:
        res['error']=repr(e)[:300]
    json.dump(res,open(out,'w'))
    print(kind,name,res.get('error') or (res['off']['stitches'],res['on']['stitches']),flush=True)
