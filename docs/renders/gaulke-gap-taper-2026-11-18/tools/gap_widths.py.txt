import sys,numpy as np,cv2
sys.path.insert(0,'.')
import gmeas
S=gmeas.S;PPM=20.0;PAD=2.0
for t in ('stroke','pull_lo','both_on'):
    d=gmeas.load(t); np.save(S+f'sw_{t}.npy',gmeas.sewn(d))
A=np.load(S+'art_off.npy')
V={k:np.load(S+f'sw_{k}.npy') for k in ('off','on','stroke','pull_lo','both_on')}
RS=[0.05*i for i in range(1,21)]
def wm(m):
    w=np.zeros(m.shape,np.float32);u=m.astype(np.uint8)
    for r in RS:
        k=cv2.getStructuringElement(cv2.MORPH_ELLIPSE,(2*int(r*PPM)+1,)*2)
        w[cv2.morphologyEx(u,cv2.MORPH_OPEN,k)>0]=2*r
    return w
def win(m,x,y,h=5.0):
    cx=int((x+PAD)*PPM);cy=int((y+PAD)*PPM);r=int(h*PPM);return m[cy-r:cy+r,cx-r:cx+r]
F=[('sun/roof gap',45,14),('peak',62,14.5),('roof band gaps',60,20),('window',57,24.5),('lower wing line',38,28.5),('G',19.5,35.5),('S',6,44),('R/E',98,35.5),('&',74,44),('PP',104,44)]
Wa=wm(~A); W={k:wm(~v) for k,v in V.items()}
print('%-16s %5s'%('white gap','art')+''.join('| %-13s'%k for k in V))
for n,x,y in F:
    a=win(~A,x,y)&(win(Wa,x,y)<=1.4)&(win(Wa,x,y)>0)
    if a.sum()<80: continue
    row='%-16s %5.2f'%(n,np.median(win(Wa,x,y)[a]))
    for k,v in V.items():
        mk=a&win(~v,x,y); row+='| %.2fmm %3.0f%%  '%(np.median(win(W[k],x,y)[mk]) if mk.any() else 0,100*mk.sum()/a.sum())
    print(row)
print('\nink area sewn/art (10x10 window) and global ink mm2:',{k:round(v.sum()/PPM**2) for k,v in V.items()},'art',round(A.sum()/PPM**2))
